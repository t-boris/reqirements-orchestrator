async def _handle_multi_ticket_approve_async(body, client: WebClient):
    """Create all tickets and show progress."""
    from src.jira.client import JiraClient
    from src.slack.channel_config import get_channel_config

    channel_id = body["channel"]["id"]
    message_ts = body["message"]["ts"]
    thread_ts = body["message"].get("thread_ts") or message_ts
    team_id = body.get("team", {}).get("id", "")

    # Get items from state
    identity = SessionIdentity(team_id=team_id, channel_id=channel_id, thread_ts=thread_ts)
    runner = get_runner(identity)
    state = await runner._get_current_state()
    multi_state = state.get("multi_ticket_state", {})
    items = multi_state.get("items", [])

    if not items:
        client.chat_postMessage(channel=channel_id, thread_ts=thread_ts, text="No items to create.")
        return

    # Get project from channel config
    config = await get_channel_config(channel_id)
    project_key = config.get("jira_project", "SCRUM")

    # Sort: Epics first, then Stories
    epics = [i for i in items if i["type"] == "epic"]
    stories = [i for i in items if i["type"] == "story"]
    ordered = epics + stories

    # Create tickets with progress updates
    results = []
    jira = JiraClient()

    for idx, item in enumerate(ordered):
        # Update preview with progress (checkmark for done, spinner for current)
        progress_blocks = build_creation_progress_blocks(ordered, results, current_idx=idx)
        client.chat_update(channel=channel_id, ts=message_ts, blocks=progress_blocks)

        try:
            # Resolve parent_item_id to Jira key if story has parent
            parent_key = None
            if item.get("parent_item_id"):
                parent_result = next((r for r in results if r["item_id"] == item["parent_item_id"]), None)
                if parent_result:
                    parent_key = parent_result["jira_key"]

            # Create ticket
            issue_type = "Epic" if item["type"] == "epic" else "Story"
            jira_key = await jira.create_issue(
                project_key=project_key,
                issue_type=issue_type,
                summary=item["title"],
                description=item.get("description", ""),
                parent_key=parent_key,
            )

            results.append({
                "item_id": item["id"],
                "jira_key": jira_key,
                "title": item["title"],
                "type": item["type"],
                "success": True,
            })
        except Exception as e:
            logger.error(f"Failed to create ticket for item {item['id']}: {e}")
            results.append({
                "item_id": item["id"],
                "title": item["title"],
                "type": item["type"],
                "success": False,
                "error": str(e),
            })

    # Final progress update
    progress_blocks = build_creation_progress_blocks(ordered, results, current_idx=None)
    client.chat_update(channel=channel_id, ts=message_ts, blocks=progress_blocks)

    # Post announcement
    await _post_creation_announcement(client, channel_id, thread_ts, results, identity)

    # Auto-track created tickets
    await _track_created_tickets(results, channel_id)

    # Clear multi_ticket_state
    await runner._update_state({"multi_ticket_state": None})
```

async def _track_created_tickets(results: list[dict], channel_id: str):
    """Auto-track all created tickets in channel."""
    from src.slack.channel_tracking import ChannelTrackingStore

    store = ChannelTrackingStore()

    for result in results:
        if result["success"]:
            await store.track_issue(
                channel_id=channel_id,
                issue_key=result["jira_key"],
                issue_type=result["type"],
                summary=result["title"],
            )

    await store.close()

async def _handle_multi_ticket_cancel_async(body, client: WebClient):
    """Cancel multi-ticket creation."""
    channel_id = body["channel"]["id"]
    message_ts = body["message"]["ts"]
    thread_ts = body["message"].get("thread_ts") or message_ts
    team_id = body.get("team", {}).get("id", "")

    # Check if any edits were made (compare to original items)
    identity = SessionIdentity(team_id=team_id, channel_id=channel_id, thread_ts=thread_ts)
    runner = get_runner(identity)
    state = await runner._get_current_state()
    multi_state = state.get("multi_ticket_state", {})

    has_edits = multi_state.get("has_edits", False)

    if has_edits:
        # Show confirmation modal
        client.views_open(
            trigger_id=body["trigger_id"],
            view={
                "type": "modal",
                "callback_id": "multi_ticket_cancel_confirm",
                "title": {"type": "plain_text", "text": "Discard Changes?"},
                "submit": {"type": "plain_text", "text": "Discard"},
                "close": {"type": "plain_text", "text": "Keep Editing"},
                "private_metadata": json.dumps({"message_ts": message_ts, "thread_ts": thread_ts, "channel_id": channel_id}),
                "blocks": [
                    {"type": "section", "text": {"type": "mrkdwn", "text": "You've made changes to the items. Are you sure you want to discard them?"}}
                ]
            }
        )
    else:
        # Direct cancel
        await _do_cancel(client, channel_id, message_ts, thread_ts, runner)

async def _do_cancel(client, channel_id, message_ts, thread_ts, runner):
    """Execute the cancel action."""
    # Delete preview message
    client.chat_delete(channel=channel_id, ts=message_ts)

    # Post dismissal
    client.chat_postMessage(
        channel=channel_id,
        thread_ts=thread_ts,
        text="Multi-ticket creation cancelled.",
    )

    # Clear state
    await runner._update_state({"multi_ticket_state": None})
```
