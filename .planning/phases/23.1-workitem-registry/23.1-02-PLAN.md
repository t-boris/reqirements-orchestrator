    CREATE TABLE IF NOT EXISTS work_items (
        id UUID PRIMARY KEY,
        channel_id TEXT NOT NULL,

        status TEXT NOT NULL DEFAULT 'draft',
        summary TEXT NOT NULL,
        description TEXT,
        facts JSONB DEFAULT '{}',

    )
""")
