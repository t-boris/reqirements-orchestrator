   """Boost score for triage-indicated mode."""
   for candidate in stage1.mode_candidates:

           candidate.score = min(1.0, candidate.score + 0.3)  # Stronger boost than constraint
   stage1.mode_candidates.sort(key=lambda c: c.score, reverse=True)
   return stage1
