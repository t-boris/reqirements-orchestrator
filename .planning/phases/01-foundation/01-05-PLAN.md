from abc import ABC, abstractmethod
import asyncpg
from src.domain.events import (
    DomainEvent, WorkItemDrafted, WorkItemProposed, WorkItemApproved,
    WorkItemCommitted, WorkItemUpdated, DecisionRecorded, DecisionProposed,
    DecisionApproved, DecisionCommitted, DecisionDeprecated
)
from src.domain.types import EntityId, EntityLifecycle
from src.domain.entities import Entity

class Projection(ABC):
    """Base for read model projections."""

    @abstractmethod
    def handles(self) -> list[str]:
        """Event types this projection handles."""
        ...

    @abstractmethod
    async def apply(self, event: DomainEvent) -> None:
        """Apply event to projection (must be idempotent)."""
        ...

class EntityProjection(Projection):
    """Projects events to current entity states."""

    def __init__(self, pool: asyncpg.Pool):
        self.pool = pool
        self.name = "entity_projection"

    def handles(self) -> list[str]:
        return [
            "WorkItemDrafted", "WorkItemProposed", "WorkItemApproved",
            "WorkItemCommitted", "WorkItemUpdated",
            "DecisionRecorded", "DecisionProposed", "DecisionApproved",
            "DecisionCommitted", "DecisionDeprecated"
        ]

    async def apply(self, event: DomainEvent) -> None:
        """Apply event to projection with upsert for idempotency."""
        match event:
            case WorkItemDrafted():
                await self._create_entity(event, EntityLifecycle.DRAFT)
            case WorkItemProposed() | DecisionProposed():
                await self._update_lifecycle(event.entity_id, EntityLifecycle.PROPOSED,
                                            canonical_message_ts=event.canonical_message_ts)
            case WorkItemApproved() | DecisionApproved():
                await self._update_lifecycle(event.entity_id, EntityLifecycle.APPROVED)
            case WorkItemCommitted():
                await self._commit_entity(event)
            case DecisionRecorded():
                await self._create_entity(event, EntityLifecycle.DRAFT)
            case DecisionCommitted():
                await self._commit_entity(event)
            case DecisionDeprecated():
                await self._deprecate_entity(event)
            case WorkItemUpdated():
                await self._update_content(event)

    async def _create_entity(self, event, lifecycle: EntityLifecycle) -> None:
        """Insert or update entity (idempotent via upsert)."""
        async with self.pool.acquire() as conn:
            await conn.execute("""
                INSERT INTO entities_view
                (id, entity_type, lifecycle, channel_id, thread_ts, content,
                 attribution, version)
                VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
                ON CONFLICT (id) DO UPDATE SET
                    content = $6, attribution = $7, version = $8, updated_at = NOW()
            """,
                str(event.entity_id),
                event.content.__class__.__name__.replace("Content", "").lower(),
                lifecycle.value,
                event.aggregate_id,
                event.thread_ts,
                event.content.model_dump(),
                {"proposed_by": event.actor_id, "proposed_at": str(event.timestamp)},
                event.version,
            )

```python
import asyncpg
from datetime import datetime
from src.infrastructure.serialization import deserialize_event
from src.infrastructure.projections import Projection

class OutboxProcessor:
    """Process outbox events and apply to projections."""

    def __init__(self, pool: asyncpg.Pool, projections: list[Projection]):
        self.pool = pool
        self.projections = projections
        self._projection_handlers: dict[str, list[Projection]] = {}
        self._build_handler_map()

    def _build_handler_map(self) -> None:
        """Build event_type -> projections map."""
        for projection in self.projections:
            for event_type in projection.handles():
                if event_type not in self._projection_handlers:
                    self._projection_handlers[event_type] = []
                self._projection_handlers[event_type].append(projection)

    async def process_pending(self, batch_size: int = 100) -> int:
        """Process pending outbox events. Returns count processed."""
        async with self.pool.acquire() as conn:
            # Get pending events (oldest first)
            rows = await conn.fetch("""
                SELECT id, event_id, event_type, payload
                FROM outbox_events
                WHERE processed_at IS NULL
                ORDER BY id ASC
                LIMIT $1
                FOR UPDATE SKIP LOCKED
            """, batch_size)

            if not rows:
                return 0

            processed = 0
            for row in rows:
                event_type = row["event_type"]
                handlers = self._projection_handlers.get(event_type, [])

                if handlers:
                    event = deserialize_event(row["payload"])
                    for projection in handlers:
                        await projection.apply(event)

                # Mark as processed
                await conn.execute("""
                    UPDATE outbox_events
                    SET processed_at = $1, status = 'processed'
                    WHERE id = $2
                """, datetime.utcnow(), row["id"])
                processed += 1

            return processed

    async def run_continuously(self, interval_ms: int = 100) -> None:
        """Run processor continuously (for background task)."""
        import asyncio
        while True:
            try:
                count = await self.process_pending()
                if count == 0:
                    await asyncio.sleep(interval_ms / 1000)
            except Exception as e:
                # Log error but continue
                await asyncio.sleep(1)  # Back off on error
```

Use FOR UPDATE SKIP LOCKED for safe concurrent processing.

@pytest.mark.asyncio
async def test_concurrency_error_on_duplicate_version(pool):
    """Test optimistic concurrency control."""
    store = EventStore(pool)
    channel_id = ChannelId(f"C{uuid4().hex[:8]}")

    event1 = WorkItemDrafted(
        aggregate_id=channel_id,
        actor_id=UserId("U123"),
        version=1,
        entity_id=EntityId.generate(),
        thread_ts=ThreadTs("1234567890.123456"),
        content=WorkItemContent(issue_type=IssueType.TASK, title="Task 1")
    )
    await store.append(event1)

    event2 = WorkItemDrafted(
        aggregate_id=channel_id,
        actor_id=UserId("U123"),
        version=1,  # Same version - should fail

        thread_ts=ThreadTs("1234567890.123457"),
        content=WorkItemContent(issue_type=IssueType.TASK, title="Task 2")

        await store.append(event2)
