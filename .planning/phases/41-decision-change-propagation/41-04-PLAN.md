# Plan 41-04: Transactional Apply + Result Card

## Objective

Create the executor that applies confirmed decision changes in truth-first order, and the result card that shows what was updated.

## Context

From 41-CONTEXT.md:
- Order of application:
  1. Update decision state in DB (new version/deprecate) — always first
  2. Update pinned decision message (Slack) — if fails, retry, but DB is already truth
  3. Update channel index pinned
  4. Jira updates: MANAGED_SECTION_ONLY, include decision_version marker
- Result card shows: Updated N, Failed M with reasons
- Each apply is idempotent (by op_id)

### Existing Infrastructure

- `src/sync/decision_sync.py` - DecisionSyncService.sync_decision()
- `src/slack/blocks/decision_cards.py` - Card builders
- `src/db/decision_store.py` - DecisionStore operations

### Task 2: DecisionChangeExecutor

Create the executor that applies changes in truth-first order.

**File:** `src/sync/decision_change_executor.py` (NEW)

**Class:** `DecisionChangeExecutor`

**Methods:**
1. `execute(op_id, skip_jira=False) -> ApplyResult`
   - Load operation and validate state is CONFIRMED
   - Transition to APPLYING
   - Execute in order: DB → Slack → Jira (if not skip_jira)
   - Track results for each phase
   - Transition to DONE or FAILED
   - Return ApplyResult

2. `_apply_to_db(op, decision) -> bool`
   - For EDIT: decision already updated when op created
   - For DEPRECATE: call decision_store.deprecate()
   - For DELETE: call decision_store.deprecate() with tombstone

3. `_apply_to_slack(op, decision, channel_id, message_ts) -> bool`
   - Update pinned decision message
   - Handle deprecated card format
   - Best effort - log failures but don't abort

4. `_apply_to_jira(op, decision) -> list[ApplyTicketResult]`
   - Use DecisionSyncService.sync_decision()
   - Collect per-ticket results
   - For DEPRECATE: add deprecation note to managed section

**Implementation:**
```python
class DecisionChangeExecutor:
    def __init__(
        self,
        decision_store: DecisionStore,
        op_store: DecisionChangeOpStore,
        link_store: DecisionLinkStore,
        jira_service: JiraService,
        slack_client: WebClient,
    ):
        self._decisions = decision_store
        self._ops = op_store
        self._links = link_store
        self._jira = jira_service
        self._slack = slack_client

    async def execute(
        self,
        op_id: str,
        skip_jira: bool = False,
    ) -> ApplyResult:
        """Execute a confirmed decision change operation.

        Truth-first ordering:
        1. DB (already updated for EDIT, needs update for DEPRECATE)
        2. Slack (best effort)
        3. Jira (if not skip_jira)
        """
        # Load operation
        op = await self._ops.get(op_id)
        if not op:
            return ApplyResult(
                op_id=op_id,
                decision_id="",
                success=False,
                error="Operation not found",
            )

        if op.state != DecisionChangeOpState.CONFIRMED:
            return ApplyResult(
                op_id=op_id,
                decision_id=op.decision_id,
                success=False,
                error=f"Operation not confirmed (state: {op.state.value})",
            )

        # Transition to APPLYING
        await self._ops.update_state(op_id, DecisionChangeOpState.APPLYING)

        result = ApplyResult(
            op_id=op_id,
            decision_id=op.decision_id,
            success=True,
        )

        try:
            # Get decision
            decision = await self._decisions.get(op.decision_id)
            if not decision:
                raise ValueError("Decision not found")

            # Phase 1: DB (for DEPRECATE/DELETE - EDIT already done)
            if op.operation in (DecisionChangeOpType.DEPRECATE, DecisionChangeOpType.DELETE):
                await self._apply_to_db(op, decision)
                decision = await self._decisions.get(op.decision_id)  # Refresh
            result.db_updated = True

            # Phase 2: Slack (best effort)
            try:
                await self._apply_to_slack(op, decision)
                result.slack_updated = True
            except Exception as e:
                logger.warning(f"Slack update failed: {e}")
                # Continue - Slack is presentation, not truth

            # Phase 3: Jira (if not skipped)
            if not skip_jira and op.impact_summary and op.impact_summary.has_jira_writes:
                ticket_results = await self._apply_to_jira(op, decision)
                result.ticket_results = ticket_results
                result.total_tickets = len(ticket_results)
                result.updated_count = sum(1 for r in ticket_results if r.success)
                result.failed_count = sum(1 for r in ticket_results if not r.success)
                result.jira_updated = result.failed_count == 0

            # Complete operation
            if result.failed_count == 0:
                await self._ops.complete(op_id)
            else:
                await self._ops.fail(op_id, f"{result.failed_count} ticket(s) failed")
                result.success = False

        except Exception as e:
            logger.error(f"Execute failed: {e}", exc_info=True)
            await self._ops.fail(op_id, str(e))
            result.success = False
            result.error = str(e)

        return result
```

### Task 3: Result Card Builder

def build_change_result_card(
    decision: Decision,
    op: DecisionChangeOp,
    result: ApplyResult,
) -> list[dict]:
    """Build result card after applying decision change."""
    blocks = []

    # Header with status
    if result.success:
        emoji = "✅"
        status = "Complete"
    else:
        emoji = "⚠️"
        status = "Completed with errors"

    blocks.append({
        "type": "header",
        "text": {"type": "plain_text", "text": f"{emoji} Decision change {status.lower()}"}
    })

    # Decision info
    blocks.append({
        "type": "section",
        "text": {"type": "mrkdwn", "text": f"*{decision.title}* (DEC-{decision.id[:8]})"}
    })

    # Results summary
    summary_lines = []
    if result.db_updated:
        summary_lines.append("✓ Database updated")
    if result.slack_updated:
        summary_lines.append("✓ Slack message updated")
    if result.jira_updated:
        summary_lines.append(f"✓ Jira updated ({result.updated_count} ticket(s))")
    elif result.total_tickets > 0:
        summary_lines.append(f"⚠️ Jira: {result.updated_count}/{result.total_tickets} updated")

    blocks.append({
        "type": "section",
        "text": {"type": "mrkdwn", "text": "\n".join(summary_lines)}
    })

    # Failed tickets details
    if result.failed_count > 0:
        failed = [r for r in result.ticket_results if not r.success]
        failed_text = "*Failed tickets:*\n"
        for r in failed[:5]:
            failed_text += f"• {r.jira_key}: {r.error or 'Unknown error'}\n"

        blocks.append({
            "type": "section",
            "text": {"type": "mrkdwn", "text": failed_text}
        })

        # Retry button
        blocks.append({
            "type": "actions",
            "elements": [
                {
                    "type": "button",
                    "text": {"type": "plain_text", "text": "Retry failed"},
                    "action_id": "decision_change_retry",
                    "value": json.dumps({"op_id": op.id}),
                },
                {
                    "type": "button",
                    "text": {"type": "plain_text", "text": "View details"},
                    "action_id": "decision_change_details",
                    "value": json.dumps({"op_id": op.id}),
                },
            ]
        })

    return blocks
```

async def _handle_decision_change_apply_async(body, client: WebClient):
    """Async handler for applying decision change with Jira sync."""
    from src.db.connection import get_connection
    from src.db.decision_change_op_store import DecisionChangeOpStore
    from src.db.decision_store import DecisionStore
    from src.db.decision_link_store import DecisionLinkStore
    from src.sync.decision_change_executor import DecisionChangeExecutor
    from src.jira.client import JiraService
    from src.config.settings import get_settings
    from src.slack.blocks.decision_cards import build_change_result_card

    # Extract data
    action = body["actions"][0]
    data = json.loads(action.get("value", "{}"))
    op_id = data.get("op_id")
    user_id = body["user"]["id"]
    channel_id = body["channel"]["id"]
    message_ts = body.get("message", {}).get("ts")

    settings = get_settings()
    jira_service = JiraService(settings)

    try:
        async with get_connection() as conn:
            op_store = DecisionChangeOpStore(conn)
            decision_store = DecisionStore(conn)
            link_store = DecisionLinkStore(conn)

            # Confirm operation
            await op_store.confirm(op_id)

            # Execute
            executor = DecisionChangeExecutor(
                decision_store=decision_store,
                op_store=op_store,
                link_store=link_store,
                jira_service=jira_service,
                slack_client=client,
            )
            result = await executor.execute(op_id, skip_jira=False)

            # Get updated decision and op
            op = await op_store.get(op_id)
            decision = await decision_store.get(op.decision_id)

        # Show result card
        result_blocks = build_change_result_card(decision, op, result)
        client.chat_update(
            channel=channel_id,
            ts=message_ts,
            blocks=result_blocks,
            text="Decision change complete" if result.success else "Decision change completed with errors",
        )

    except Exception as e:
        logger.error(f"Apply failed: {e}", exc_info=True)
        client.chat_postEphemeral(
            channel=channel_id,
            user=user_id,
            text=f"Failed to apply changes: {str(e)}",
        )
    finally:
        await jira_service.close()
```
