# Plan 41-05: Rollback Support + Error Recovery

## Objective

Add retry and rollback capabilities to decision change operations. When Jira updates fail, users can retry or rollback the changes.

## Context

From 41-CONTEXT.md:
- Result card shows: [Retry failed] [Rollback Jira patches] [Open diff]
- Failed updates can be retried
- Rollback = revert Jira managed sections to previous decision version
- For deprecate: Jira managed section shows "Decision deprecated, see successor"

### Key Scenarios

1. **Partial failure**: 5/7 tickets updated, 2 failed → show [Retry failed]
2. **All failed**: Network error → show [Retry all]
3. **User regret**: All updated but user wants to undo → show [Rollback]
4. **Deprecation**: Need to update Jira with deprecation notice

```python
async def _handle_decision_change_retry_async(body, client: WebClient):
    """Retry failed Jira updates for a decision change operation."""
    from src.db.connection import get_connection
    from src.db.decision_change_op_store import DecisionChangeOpStore
    from src.db.decision_store import DecisionStore
    from src.db.decision_link_store import DecisionLinkStore
    from src.sync.decision_change_executor import DecisionChangeExecutor
    from src.jira.client import JiraService
    from src.config.settings import get_settings
    from src.slack.blocks.decision_cards import build_change_result_card

    action = body["actions"][0]
    data = json.loads(action.get("value", "{}"))
    op_id = data.get("op_id")
    user_id = body["user"]["id"]
    channel_id = body["channel"]["id"]
    message_ts = body.get("message", {}).get("ts")

    settings = get_settings()
    jira_service = JiraService(settings)

    try:
        async with get_connection() as conn:
            op_store = DecisionChangeOpStore(conn)
            decision_store = DecisionStore(conn)
            link_store = DecisionLinkStore(conn)

            op = await op_store.get(op_id)
            if not op:
                client.chat_postEphemeral(
                    channel=channel_id,
                    user=user_id,
                    text="Operation not found.",
                )
                return

            # Must be in FAILED state to retry
            if op.state != DecisionChangeOpState.FAILED:
                client.chat_postEphemeral(
                    channel=channel_id,
                    user=user_id,
                    text=f"Cannot retry operation in {op.state.value} state.",
                )
                return

            # Reset to CONFIRMED for retry
            await op_store.update_state(op_id, DecisionChangeOpState.CONFIRMED)

            # Re-execute
            executor = DecisionChangeExecutor(
                decision_store=decision_store,
                op_store=op_store,
                link_store=link_store,
                jira_service=jira_service,
                slack_client=client,
            )
            result = await executor.execute(op_id)

            # Get updated decision and op
            op = await op_store.get(op_id)
            decision = await decision_store.get(op.decision_id)

        # Update result card
        result_blocks = build_change_result_card(decision, op, result)
        client.chat_update(
            channel=channel_id,
            ts=message_ts,
            blocks=result_blocks,
            text="Retry complete" if result.success else "Retry completed with errors",
        )

    except Exception as e:
        logger.error(f"Retry failed: {e}", exc_info=True)
        client.chat_postEphemeral(
            channel=channel_id,
            user=user_id,
            text=f"Retry failed: {str(e)}",
        )
    finally:
        await jira_service.close()
```

### Task 2: Rollback Service

Create a service to rollback Jira managed sections to previous state.

**File:** `src/sync/decision_rollback.py` (NEW)

**Class:** `DecisionRollbackService`

**Methods:**
1. `rollback(op_id) -> RollbackResult`
   - Get operation and its previous state
   - For EDIT: sync previous version to Jira
   - For DEPRECATE: remove deprecation notice, restore original
   - Track rollback in operation history

**Implementation:**
```python
@dataclass
class RollbackResult:
    success: bool
    rolled_back_count: int = 0
    failed_count: int = 0
    error: Optional[str] = None


class DecisionRollbackService:
    def __init__(
        self,
        decision_store: DecisionStore,
        op_store: DecisionChangeOpStore,
        link_store: DecisionLinkStore,
        jira_service: JiraService,
    ):
        self._decisions = decision_store
        self._ops = op_store
        self._links = link_store
        self._jira = jira_service

    async def rollback(self, op_id: str) -> RollbackResult:
        """Rollback a completed decision change operation.

        For EDIT: Reverts Jira managed sections to from_version content.
        For DEPRECATE: Removes deprecation notice, restores decision.

        Note: DB state is NOT rolled back. The decision remains at current
        version. Only Jira projection is reverted.
        """
        op = await self._ops.get(op_id)
        if not op:
            return RollbackResult(success=False, error="Operation not found")

        if op.state != DecisionChangeOpState.DONE:
            return RollbackResult(
                success=False,
                error=f"Can only rollback DONE operations, got {op.state.value}"
            )

        decision = await self._decisions.get(op.decision_id)
        if not decision:
            return RollbackResult(success=False, error="Decision not found")

        # Get version history to find previous state
        history = await self._decisions.get_version_history(op.decision_id)
        prev_version = next(
            (v for v in history if v.version == op.from_version),
            None
        )

        if op.operation == DecisionChangeOpType.EDIT and prev_version:
            # Sync previous version to Jira
            return await self._rollback_edit(decision, prev_version)
        elif op.operation == DecisionChangeOpType.DEPRECATE:
            # Remove deprecation notice (but decision stays deprecated in DB)
            return await self._rollback_deprecate(decision)
        else:
            return RollbackResult(
                success=False,
                error="Cannot rollback this operation type"
            )

    async def _rollback_edit(
        self,
        decision: Decision,
        prev_version,  # DecisionVersion
    ) -> RollbackResult:
        """Rollback an edit by syncing previous version to Jira."""
        links = await self._links.get_links_for_decision(decision.id)
        rolled_back = 0
        failed = 0

        for link in links:
            try:
                # Get current Jira description
                issue = await self._jira.get_issue(link.jira_key)
                current_desc = issue.description or ""

                # Update managed section with previous version content
                new_desc = update_description_with_decision(
                    current_desc,
                    prev_version.title,
                    prev_version.description,
                    prev_version.version,
                    rollback=True,
                )

                await self._jira.update_issue(
                    issue_key=link.jira_key,
                    updates={"description": new_desc},
                )
                rolled_back += 1
            except Exception as e:
                logger.error(f"Rollback failed for {link.jira_key}: {e}")
                failed += 1

        return RollbackResult(
            success=failed == 0,
            rolled_back_count=rolled_back,
            failed_count=failed,
        )
```

```python
def build_decision_section(
    decision: Decision,
    deprecated: bool = False,
    successor_id: Optional[str] = None,
) -> str:
    """Build the decision section for Jira managed content.

    Args:
        decision: The decision to render
        deprecated: Whether to show deprecation notice
        successor_id: ID of successor decision (if deprecated)
    """
    if deprecated:
        lines = [
            f"**Decision {decision.title}** (DEC-{decision.id[:8]})",
            "",
            "⚠️ **This decision has been deprecated.**",
        ]
        if successor_id:
            lines.append(f"See replacement: DEC-{successor_id[:8]}")
        if decision.deprecated_reason:
            lines.append(f"Reason: {decision.deprecated_reason}")
        return "\n".join(lines)

    # Normal decision format
    lines = [
        f"**Decision {decision.title}** (DEC-{decision.id[:8]} v{decision.version})",
        "",
        decision.description,
    ]

    # Add rich context if available
    if decision.rationale:
        lines.append("")
        lines.append("**Rationale:**")
        for item in decision.rationale[:3]:
            lines.append(f"- {item.text}")

    return "\n".join(lines)
```
