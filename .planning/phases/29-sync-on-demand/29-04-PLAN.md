```python
async def _handle_ticket_action_with_preflight(
    client: WebClient,
    channel_id: str,
    user_id: str,
    result: dict,
    thread_ts: str | None,
) -> None:
    """Handle ticket action with preflight check.

    Before executing ticket action, check for conflicts:
    1. Fetch current Jira state
    2. Compare with local state
    3. Classify conflict
    4. If IDEMPOTENT: Auto-succeed
    5. If SAFE_DRIFT: Show warning, default proceed
    6. If REAL_CONFLICT: Block, require choice
    7. If STRUCTURAL: Block, explain why
    """
    action_type = result.get("action_type")
    ticket_key = result.get("ticket_key")

    # Get preflight service
    settings = get_settings()
    jira = JiraService(settings)
    async with get_connection() as conn:
        registry = JiraRegistryStore(conn)
        preflight = PreflightService(jira, registry)

        if action_type == "transition":
            # Check transition preflight
            target_status = result.get("target_status")
            preflight_result = await preflight.check_transition(
                channel_id, ticket_key, target_status
            )
        elif action_type == "update":
            # Check update preflight
            fields = result.get("fields", {})
            preflight_result = await preflight.check_update(
                channel_id, ticket_key, fields
            )
        else:
            preflight_result = None

        if preflight_result and preflight_result.needs_choice:
            # Show preflight UI, don't execute
            blocks = build_preflight_blocks(preflight_result)
            await client.chat_postMessage(
                channel=channel_id,
                thread_ts=thread_ts,
                blocks=blocks,
            )
            return

        # Proceed with action (IDEMPOTENT auto-succeeds, no conflict proceeds)
        await _execute_ticket_action(...)
```

```python
async def handle_preflight_button(
    client: WebClient,
    payload: dict,
) -> None:
    """Handle preflight button clicks.

    Button action_ids:
    - preflight_proceed: Execute pending action
    - preflight_pull_only: Sync from Jira, abort action
    - preflight_use_jira: Resolve conflict using Jira values
    - preflight_use_channel: Resolve conflict using local values
    - preflight_cancel: Abort, do nothing

    Button value contains JSON with:
    - channel_id
    - jira_key
    - pending_action: dict with action details
    - conflict_type: for context
    """
    action_id = payload["actions"][0]["action_id"]
    value = json.loads(payload["actions"][0]["value"])

    channel_id = value["channel_id"]
    jira_key = value["jira_key"]
    pending_action = value.get("pending_action", {})

    if action_id == "preflight_proceed":
        # Execute the pending action
        await _execute_pending_action(client, channel_id, pending_action)
        # Update message to show success
        await _update_preflight_message(client, payload, "Action completed")

    elif action_id == "preflight_pull_only":
        # Sync from Jira, don't execute
        # Update registry with fresh Jira data
        await _sync_from_jira(channel_id, jira_key)
        await _update_preflight_message(client, payload, "Synced from Jira")

    elif action_id == "preflight_use_jira":
        # Update local state to match Jira, then execute action
        await _resolve_conflict_use_jira(channel_id, jira_key)
        await _execute_pending_action(client, channel_id, pending_action)
        await _update_preflight_message(client, payload, "Resolved using Jira version")

    elif action_id == "preflight_use_channel":
        # Force local version, execute action (override Jira)
        await _execute_pending_action(client, channel_id, pending_action, force=True)
        await _update_preflight_message(client, payload, "Applied local version")

    elif action_id == "preflight_cancel":
        # Just update the message
        await _update_preflight_message(client, payload, "Operation cancelled")
```
