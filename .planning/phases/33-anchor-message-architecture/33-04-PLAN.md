```python
@dataclass
class ThreadContext:
    """Resolved context for a thread.

    Represents what object a thread is managing.
    This is the result of context resolution (Rule A3).
    """
    anchor_type: AnchorType
    object_id: str
    anchor_message_ts: str  # The anchor message that created this context

    # Hydrated entity (optional, for convenience)
    decision: Optional[Decision] = None
    workitem: Optional[WorkItem] = None
    jira_key: Optional[str] = None  # Legacy binding

    @property
    def has_entity(self) -> bool:
        """Check if any entity is hydrated."""
        return self.decision is not None or self.workitem is not None

    @property
    def display_id(self) -> str:
        """Human-readable identifier for the context."""
        if self.anchor_type == AnchorType.DECISION:
            return f"DEC-{self.object_id[:8]}"
        elif self.anchor_type == AnchorType.WORKITEM:
            return self.jira_key or f"WI-{self.object_id[:8]}"
        return self.object_id
```

```python
"""Thread context resolution service.

Implements Rule A3: Context Inheritance.
Messages in a thread automatically inherit the object_id of the anchor.
"""

class ContextResolver:
    """Resolves what object a thread is managing.

    Resolution order (first match wins):
    1. AnchorStore - check if thread_ts is an anchor message
    2. DecisionStore - check for decision canonical message
    3. WorkItemStore - check for workitem canonical message
    4. ThreadBindingStore - legacy Jira key binding

    This multi-tier approach ensures backward compatibility while
    supporting the new anchor-based pattern.
    """

    def __init__(
        self,
        anchor_store: AnchorStore,
        decision_store: DecisionStore,
        workitem_store: WorkItemStore,
        binding_store: ThreadBindingStore
    ):
        self._anchor_store = anchor_store
        self._decision_store = decision_store
        self._workitem_store = workitem_store
        self._binding_store = binding_store

    async def resolve(
        self,
        channel_id: str,
        thread_ts: str,
        hydrate: bool = False
    ) -> Optional[ThreadContext]:
        """Resolve thread to its managing object.

        Args:
            channel_id: Slack channel ID
            thread_ts: Thread timestamp (parent message)
            hydrate: If True, load full entity into context

        Returns:
            ThreadContext if thread is anchored to an object, None otherwise
        """
        # 1. Check AnchorStore (new pattern)
        anchor = await self._anchor_store.get_by_message(channel_id, thread_ts)
        if anchor:
            ctx = ThreadContext(
                anchor_type=anchor.anchor_type,
                object_id=anchor.object_id,
                anchor_message_ts=anchor.message_ts
            )
            if hydrate:
                await self._hydrate_context(ctx)
            return ctx

        # 2. Check DecisionStore (existing pattern)
        decision = await self._decision_store.get_by_canonical_message(channel_id, thread_ts)
        if decision:
            return ThreadContext(
                anchor_type=AnchorType.DECISION,
                object_id=str(decision.id),
                anchor_message_ts=thread_ts,
                decision=decision if hydrate else None
            )

        # 3. Check WorkItemStore (after 33-03)
        workitem = await self._workitem_store.get_by_canonical_message(channel_id, thread_ts)
        if workitem:
            return ThreadContext(
                anchor_type=AnchorType.WORKITEM,
                object_id=str(workitem.id),
                anchor_message_ts=thread_ts,
                workitem=workitem if hydrate else None,
                jira_key=workitem.jira_key
            )

        # 4. Check legacy ThreadBindingStore
        binding = await self._binding_store.get_binding(channel_id, thread_ts)
        if binding:
            return ThreadContext(
                anchor_type=AnchorType.WORKITEM,  # Legacy bindings are Jira-based
                object_id=binding.issue_key,
                anchor_message_ts=thread_ts,
                jira_key=binding.issue_key
            )

        return None

    async def _hydrate_context(self, ctx: ThreadContext) -> None:
        """Load full entity into context."""
        if ctx.anchor_type == AnchorType.DECISION:
            ctx.decision = await self._decision_store.get(UUID(ctx.object_id))
        elif ctx.anchor_type == AnchorType.WORKITEM:
            ctx.workitem = await self._workitem_store.get(UUID(ctx.object_id))
            if ctx.workitem:
                ctx.jira_key = ctx.workitem.jira_key


# Singleton instance
_resolver: Optional[ContextResolver] = None

def get_context_resolver() -> ContextResolver:
    """Get the singleton ContextResolver instance."""
    global _resolver
    if _resolver is None:
        raise RuntimeError("ContextResolver not initialized. Call initialize_resolver first.")
    return _resolver

async def initialize_resolver(
    anchor_store: AnchorStore,
    decision_store: DecisionStore,
    workitem_store: WorkItemStore,
    binding_store: ThreadBindingStore
) -> ContextResolver:
    """Initialize the singleton ContextResolver."""
    global _resolver
    _resolver = ContextResolver(anchor_store, decision_store, workitem_store, binding_store)
    return _resolver
```
