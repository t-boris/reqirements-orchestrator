```python
async def _resolve_contextual_target(
    state: AgentState,
    channel_id: str,
    thread_ts: Optional[str]
) -> Optional[str]:
    """Resolve the target Jira issue for a command.

    Resolution order (Rule A4: Implicit Commands):
    1. Explicit mention in message (SCRUM-123)
    2. thread_context from AgentState (anchor resolution)
    3. Legacy thread binding
    4. Channel tracker (most recent issue)

    The key change is step 2: if we're in an anchored thread,
    the anchor's object is the default target.
    """
    # 1. Check for explicit mention in message
    explicit_key = _extract_explicit_jira_key(state.get("message", ""))
    if explicit_key:
        logger.debug(f"Using explicit Jira key: {explicit_key}")
        return explicit_key

    # 2. NEW: Check thread_context from anchor resolution (Rule A4)
    thread_context = state.get("thread_context")
    if thread_context:
        if thread_context.jira_key:
            logger.debug(f"Using anchor context Jira key: {thread_context.jira_key}")
            return thread_context.jira_key
        # If anchor is for WorkItem without Jira key yet, return workitem_id
        if thread_context.anchor_type == AnchorType.WORKITEM:
            logger.debug(f"Using anchor context WorkItem: {thread_context.object_id}")
            # Caller handles WorkItem lookup
            return f"WI:{thread_context.object_id}"

    # 3. Legacy thread binding (fallback)
    if thread_ts:
        binding_store = ThreadBindingStore.get_instance()
        binding = await binding_store.get_binding(channel_id, thread_ts)
        if binding:
            logger.debug(f"Using legacy binding: {binding.issue_key}")
            return binding.issue_key

    # 4. Channel tracker fallback
    tracker = ChannelTracker.get_instance()
    recent = tracker.get_recent_issues(channel_id, limit=1)
    if recent:
        logger.debug(f"Using channel tracker: {recent[0]}")
        return recent[0]

    return None
```

```python
def _get_implicit_command_patterns() -> list[tuple[str, str]]:
    """Patterns that imply action on thread's anchored object.

    These commands don't need explicit "SCRUM-123" when in anchor thread.
    """
    return [
        # Update commands
        (r"update (?:the )?description", "TICKET_ACTION"),
        (r"change (?:the )?(?:title|summary)", "TICKET_ACTION"),
        (r"add (?:a )?comment", "TICKET_ACTION"),

        # Status commands
        (r"mark (?:as )?(?:done|complete|finished)", "TICKET_ACTION"),
        (r"move to (?:in progress|done|review)", "TICKET_ACTION"),
        (r"assign (?:to|this)", "TICKET_ACTION"),

        # WorkItem commands
        (r"add (?:a )?story", "WORKITEM_CREATE"),
        (r"split into stories", "DRAFT_TRANSFORM"),
        (r"decompose", "DRAFT_TRANSFORM"),

        # Decision commands
        (r"deprecate(?: this)?", "DECISION"),
        (r"update decision", "DECISION"),
    ]

async def classify_intent_with_context(
    message: str,
    thread_context: Optional[ThreadContext]
) -> Intent:
    """Classify intent with thread context awareness.

    If in anchored thread, implicit commands are recognized.
    """
    # First, check implicit command patterns if we have context
    if thread_context:
        for pattern, intent in _get_implicit_command_patterns():
            if re.search(pattern, message, re.IGNORECASE):
                logger.debug(f"Implicit command '{pattern}' in anchored thread -> {intent}")
                return Intent(intent)

    # Fall through to regular classification
    return await classify_intent(message)
```

```python
async def _dispatch_with_context(
    state: AgentState,
    thread_context: Optional[ThreadContext]
) -> DispatchResult:
    """Dispatch message with thread context awareness.

    When thread has context, use context-aware classification
    and inject implicit target into state.
    """
    message = state.get("message", "")

    # Use context-aware classification
    intent = await classify_intent_with_context(message, thread_context)

    # If we have thread context and no explicit target, inject it
    if thread_context and not _has_explicit_target(message):
        # Add implicit target to state for downstream use
        state["implicit_target"] = thread_context.object_id
        state["implicit_target_type"] = thread_context.anchor_type.value

        # Log for debugging
        logger.info(
            f"Implicit target from anchor: {thread_context.display_id} "
            f"(type={thread_context.anchor_type.value})"
        )

    return await dispatch(state, intent)
```
