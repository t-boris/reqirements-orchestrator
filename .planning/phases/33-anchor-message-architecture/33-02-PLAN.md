   - Add `_pool: Optional[Pool]` for database connection
   - Add `async def initialize(self, pool: Pool)` method

3. Implementation pattern (follow WorkItemStore):

```python
class ThreadBindingStore:
    """Database-backed store for thread-to-Jira bindings.

    Persists bindings so context survives bot restarts.
    Legacy store: binds threads to Jira keys.
    For entity-based bindings, see AnchorStore.
    """
    _instance: Optional["ThreadBindingStore"] = None
    _pool: Optional[Pool] = None

    @classmethod
    def get_instance(cls) -> "ThreadBindingStore":
        if cls._instance is None:
            cls._instance = cls()
        return cls._instance

    async def initialize(self, pool: Pool) -> None:
        self._pool = pool

    async def bind(
        self,
        channel_id: str,
        thread_ts: str,
        issue_key: str,
        bound_by: str
    ) -> None:
        """Create or update thread binding."""
        async with self._pool.acquire() as conn:
            await conn.execute('''
                INSERT INTO thread_bindings (channel_id, thread_ts, issue_key, bound_by)
                VALUES ($1, $2, $3, $4)
                ON CONFLICT (channel_id, thread_ts)
                DO UPDATE SET issue_key = $3, bound_at = NOW()
            ''', channel_id, thread_ts, issue_key, bound_by)

    async def get_binding(
        self,
        channel_id: str,
        thread_ts: str
    ) -> Optional[ThreadBinding]:
        """Get binding for thread."""
        async with self._pool.acquire() as conn:
            row = await conn.fetchrow('''
                SELECT channel_id, thread_ts, issue_key, bound_at, bound_by
                FROM thread_bindings
                WHERE channel_id = $1 AND thread_ts = $2
            ''', channel_id, thread_ts)
            if row:
                return ThreadBinding(
                    channel_id=row['channel_id'],
                    thread_ts=row['thread_ts'],
                    issue_key=row['issue_key'],
                    bound_at=row['bound_at'],
                    bound_by=row['bound_by']
                )
            return None

    async def unbind(
        self,
        channel_id: str,
        thread_ts: str
    ) -> bool:
        """Remove thread binding."""
        async with self._pool.acquire() as conn:
            result = await conn.execute('''
                DELETE FROM thread_bindings
                WHERE channel_id = $1 AND thread_ts = $2
            ''', channel_id, thread_ts)
            return result != "DELETE 0"
```
