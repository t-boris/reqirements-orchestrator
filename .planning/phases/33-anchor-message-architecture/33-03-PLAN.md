async def set_canonical_message(
    self,
    workitem_id: UUID,
    channel_id: str,
    message_ts: str
) -> bool:
    """Set the canonical message for a WorkItem.

    Called after posting the anchor message to Slack.
    Also creates entry in AnchorStore for reverse lookup.
    """
    async with self._pool.acquire() as conn:
        result = await conn.execute('''
            UPDATE work_items
            SET canonical_message_ts = $2,
                canonical_channel_id = $3,
                updated_at = NOW()
            WHERE id = $1
        ''', workitem_id, message_ts, channel_id)

        # Also register in AnchorStore for reverse lookup
        anchor_store = AnchorStore.get_instance()
        await anchor_store.create_anchor(
            anchor_type=AnchorType.WORKITEM,
            object_id=str(workitem_id),
            channel_id=channel_id,
            message_ts=message_ts,
            created_by="system"  # Will be updated with actual user
        )

        return result != "UPDATE 0"

async def get_by_canonical_message(
    self,
    channel_id: str,
    message_ts: str
) -> Optional[WorkItem]:
    """Lookup WorkItem by its canonical message.

    Use case: When user posts in thread under WorkItem anchor.
    """
    async with self._pool.acquire() as conn:
        row = await conn.fetchrow('''
            SELECT * FROM work_items
            WHERE canonical_channel_id = $1
              AND canonical_message_ts = $2
        ''', channel_id, message_ts)
        if row:
            return WorkItem(**dict(row))
        return None
```

def build_workitem_anchor_blocks(
    workitem: WorkItem,
    jira_key: Optional[str] = None
) -> list[dict]:
    """Build blocks for WorkItem anchor message.

    The anchor message is the canonical representation of the WorkItem.
    Format follows Decision anchor pattern:
    - Type emoji + ID + title
    - Status indicator
    - Action buttons
    """
    # Type emoji mapping
    type_emoji = {
        "epic": ":large_purple_circle:",
        "story": ":large_blue_circle:",
        "task": ":white_circle:",
        "bug": ":red_circle:",
    }.get(workitem.issue_type.lower() if workitem.issue_type else "task", ":white_circle:")

    # Status emoji
    status_emoji = {
        "draft": ":pencil2:",
        "approved": ":white_check_mark:",
        "committed": ":rocket:",
    }.get(workitem.status.lower(), ":grey_question:")

    # Build header
    title = workitem.summary or workitem.title or "Untitled"
    header = f"{type_emoji} *{title}*"
    if jira_key:
        header = f"{type_emoji} *{jira_key}* — {title}"

    blocks = [
        {
            "type": "section",
            "text": {"type": "mrkdwn", "text": header}
        }
    ]

    # Status line
    status_text = f"{status_emoji} Status: {workitem.status}"
    if workitem.assignee:
        status_text += f" | Assignee: <@{workitem.assignee}>"

    blocks.append({
        "type": "context",
        "elements": [{"type": "mrkdwn", "text": status_text}]
    })

    return blocks
```
