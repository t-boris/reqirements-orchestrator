from enum import Enum
from typing import Optional
from pydantic import BaseModel


class OpsSubtype(str, Enum):
    """Subtypes for OPS intent."""
    DEBUG = "debug"      # Triage failures, retry
    EXPLAIN = "explain"  # Policy trace, show reasoning


class Intent(str, Enum):
    """Intent classification types."""
    # Primary intents
    WORKITEM_CREATE = "workitem_create"  # Create new work item
    TICKET_ACTION = "ticket_action"      # Actions on existing ticket
    JIRA_COMMAND = "jira_command"        # Modify ticket fields
    SYNC_REQUEST = "sync_request"        # Bulk sync
    JIRA_SEARCH = "jira_search"          # Search existing
    REVIEW = "review"                    # Analysis/feedback
    DISCUSSION = "discussion"            # Greeting/casual
    META = "meta"                        # Questions about bot
    AMBIGUOUS = "ambiguous"              # Unclear intent
    CHANGE_REQUEST = "change_request"    # Diff-based updates (Phase 25.3)
    OPS = "ops"                          # Operational mode (debug/explain)

    # Deprecated aliases
    TICKET = "ticket"  # @deprecated: Use WORKITEM_CREATE

    @classmethod
    def normalize(cls, intent: "Intent") -> "Intent":
        """Normalize deprecated intents to their current equivalents."""
        if intent == cls.TICKET:
            return cls.WORKITEM_CREATE
        return intent


class IntentResult(BaseModel):
    """Result of intent classification."""
    intent: Intent
    confidence: float
    persona: Optional[str] = None
    ticket_key: Optional[str] = None
    action_type: Optional[str] = None
    command_type: Optional[str] = None
    command_field: Optional[str] = None
    command_value: Optional[str] = None
    search_query: Optional[str] = None
    reason: Optional[str] = None
    # New for OPS intent
    ops_subtype: Optional[OpsSubtype] = None

```python
"""OPS intent handler - operational diagnostics and explanations."""

import logging
from typing import Any

from src.schemas.intent import OpsSubtype
from src.graph.state import AgentState

logger = logging.getLogger(__name__)


async def ops_node(state: AgentState) -> dict[str, Any]:
    """Handle OPS intent with DEBUG or EXPLAIN subtype.

    DEBUG: Triage failures, show what went wrong, suggest retry
    EXPLAIN: Show policy trace, explain decision reasoning
    """
    subtype = state.get("ops_subtype", OpsSubtype.DEBUG)
    channel_id = state["channel_id"]
    thread_ts = state.get("thread_ts")
    message = state.get("user_message", "")

    if subtype == OpsSubtype.DEBUG:
        return await _handle_debug(state, channel_id, thread_ts, message)
    else:
        return await _handle_explain(state, channel_id, thread_ts, message)


async def _handle_debug(
    state: AgentState,
    channel_id: str,
    thread_ts: str | None,
    message: str,
) -> dict[str, Any]:
    """Handle DEBUG subtype - triage failures.

    Flow:
    1. Identify what failed (last error in thread/channel)
    2. Analyze probable cause
    3. Suggest fix or retry plan
    """
    # Get recent errors from context
    recent_errors = state.get("recent_errors", [])
    last_action = state.get("last_action")

    # Build triage response
    response_parts = ["*Investigating failure...*\n"]

    if recent_errors:
        response_parts.append("*What failed:*")
        for err in recent_errors[-3:]:  # Last 3 errors
            response_parts.append(f"- {err.get('summary', 'Unknown error')}")
        response_parts.append("")

    if last_action:
        response_parts.append(f"*Last action:* {last_action}")
        response_parts.append("")

    # Analyze and suggest
    response_parts.append("*Probable cause:* Analyzing...")
    response_parts.append("*Suggested action:* Retry or check configuration")

    return {
        "action": "ops_response",
        "ops_subtype": "debug",
        "response": "\n".join(response_parts),
        "show_retry_button": bool(recent_errors),
    }


async def _handle_explain(
    state: AgentState,
    channel_id: str,
    thread_ts: str | None,
    message: str,
) -> dict[str, Any]:
    """Handle EXPLAIN subtype - show decision reasoning.

    Output in state machine terms:
    - Intent detected: X
    - Policy applied: Y
    - Action executed: Z
    """
    # Get decision trace from state
    last_intent = state.get("last_intent")
    last_decision = state.get("last_decision")
    last_action = state.get("last_action")

    response_parts = ["*Decision Trace*\n"]

    if last_intent:
        response_parts.append(f"*Intent detected:* `{last_intent}`")

    if last_decision:
        response_parts.append(f"*Policy applied:* {last_decision.get('policy', 'default')}")
        if last_decision.get("reason"):
            response_parts.append(f"*Reason:* {last_decision['reason']}")

    if last_action:
        response_parts.append(f"*Action executed:* `{last_action}`")

    if not any([last_intent, last_decision, last_action]):
        response_parts.append("_No recent decisions to explain._")
        response_parts.append("_Try asking after I've processed a message._")

    return {
        "action": "ops_response",
        "ops_subtype": "explain",
        "response": "\n".join(response_parts),
    }
```
