   - MENTION — User @mentioned bot
   - COMMAND — /command invoked

   - TIMEOUT — 10min without user response
   - USER_CANCEL — User explicitly canceled

   - channel_id: str
   - thread_ts: str
   - mode: ConversationMode (default PASSIVE)
   - entered_at: datetime
   - last_activity: datetime
   - unanswered_questions: int = 0  # For budget tracking
   - transition_reason: Optional[ModeTransitionReason]

4. Helper functions:
   ```python
   def should_activate(reason: ModeTransitionReason) -> bool:
       """Check if reason should transition to ACTIVE."""
       return reason in (ModeTransitionReason.MENTION, ModeTransitionReason.COMMAND, ModeTransitionReason.TASK_BLOCKED)

   def should_deactivate(reason: ModeTransitionReason) -> bool:
       """Check if reason should transition to PASSIVE."""
       return reason in (ModeTransitionReason.PLAN_COMPLETE, ModeTransitionReason.TIMEOUT, ModeTransitionReason.USER_CANCEL)
   ```

   ```python
   class ConversationModeStore:
       def __init__(self, conn: AsyncConnection) -> None:
           self._conn = conn

       async def create_tables(self) -> None:
           """Create conversation_modes table."""

       async def get(self, channel_id: str, thread_ts: str) -> Optional[ConversationModeState]:
           """Get current mode state for thread."""

       async def get_or_create(self, channel_id: str, thread_ts: str) -> ConversationModeState:
           """Get or create mode state (defaults to PASSIVE)."""

       async def transition(
           self,
           channel_id: str,
           thread_ts: str,
           new_mode: ConversationMode,
           reason: ModeTransitionReason,
       ) -> ConversationModeState:
           """Transition to new mode with reason."""

       async def increment_unanswered(self, channel_id: str, thread_ts: str) -> int:
           """Increment unanswered question count, return new value."""

       async def reset_unanswered(self, channel_id: str, thread_ts: str) -> None:
           """Reset unanswered count to 0 (on new user message)."""

       async def update_activity(self, channel_id: str, thread_ts: str) -> None:
           """Update last_activity timestamp."""
   ```

2. Table schema:
   ```sql
   CREATE TABLE IF NOT EXISTS conversation_modes (
       channel_id TEXT NOT NULL,
       thread_ts TEXT NOT NULL,
       mode TEXT NOT NULL DEFAULT 'passive',
       entered_at TIMESTAMPTZ NOT NULL,
       last_activity TIMESTAMPTZ NOT NULL,
       unanswered_questions INTEGER NOT NULL DEFAULT 0,
       transition_reason TEXT,
       PRIMARY KEY (channel_id, thread_ts)
   )
   ```
