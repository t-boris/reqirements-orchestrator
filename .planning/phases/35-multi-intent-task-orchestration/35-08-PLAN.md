```python
async def _handle_task_plan_created(
    client: WebClient,
    state: AgentState,
    session: SessionIdentity,
) -> None:
    """Handle newly created TaskPlan - post status card and announcement."""
    task_plan_data = state.get("task_plan")
    if not task_plan_data:
        return

    task_plan = TaskPlan.model_validate(task_plan_data)

    # Post announcement message
    announcement = build_multi_intent_announcement(task_plan)
    await client.chat_postMessage(
        channel=session.channel_id,
        thread_ts=session.thread_ts,
        text=announcement,
    )

    # Post status card
    updater = TaskStatusUpdater(client)
    await updater.post_initial_card(
        task_plan,
        session.channel_id,
        session.thread_ts,
    )

    logger.info(f"Posted TaskPlan {task_plan.plan_id} with {len(task_plan.tasks)} tasks")


async def _handle_task_confirmation(
    client: WebClient,
    state: AgentState,
    session: SessionIdentity,
    decision_result: dict,
) -> None:
    """Handle task needing confirmation - update status card."""
    task_plan_data = state.get("task_plan")
    if not task_plan_data:
        return

    task_plan = TaskPlan.model_validate(task_plan_data)

    # Update status card to show blocked task
    updater = TaskStatusUpdater(client)
    await updater.update_card(
        task_plan,
        session.channel_id,
        event="task_blocked",
    )

    # Post reminder about which task needs approval
    task_title = decision_result.get("task_title", "Unknown task")
    await client.chat_postMessage(
        channel=session.channel_id,
        thread_ts=session.thread_ts,
        text=f"⏸️ Waiting for approval: *{task_title}*\nUse the buttons above to approve or reject.",
    )


async def _handle_task_plan_complete(
    client: WebClient,
    state: AgentState,
    session: SessionIdentity,
) -> None:
    """Handle completed TaskPlan - post summary to channel."""
    task_plan_data = state.get("task_plan")
    if not task_plan_data:
        return

    task_plan = TaskPlan.model_validate(task_plan_data)

    # Final status card update
    updater = TaskStatusUpdater(client)
    await updater.flush_pending(task_plan.plan_id, session.channel_id)
    await updater.update_card(
        task_plan,
        session.channel_id,
        event="plan_completed",
    )

    # Post completion summary to channel (not just thread)
    summary = build_plan_complete_message(task_plan)
    await client.chat_postMessage(
        channel=session.channel_id,
        text=summary,
    )

    logger.info(f"TaskPlan {task_plan.plan_id} completed")


async def _handle_task_plan_blocked(
    client: WebClient,
    state: AgentState,
    session: SessionIdentity,
) -> None:
    """Handle TaskPlan that's blocked waiting for user input."""
    task_plan_data = state.get("task_plan")
    if not task_plan_data:
        return

    task_plan = TaskPlan.model_validate(task_plan_data)

    # Update status card
    updater = TaskStatusUpdater(client)
    await updater.update_card(
        task_plan,
        session.channel_id,
        event="task_blocked",
    )

    # Find blocked tasks
    blocked = [t for t in task_plan.tasks if t.status == TaskStatus.BLOCKED]
    if blocked:
        task_names = ", ".join(t.title for t in blocked[:3])
        if len(blocked) > 3:
            task_names += f", and {len(blocked) - 3} more"
        await client.chat_postMessage(
            channel=session.channel_id,
            thread_ts=session.thread_ts,
            text=f"⏸️ Waiting for: {task_names}",
        )
```

```python
async def intent_router_node(state: AgentState) -> dict[str, Any]:
    """Route based on intent classification.

    Phase 35: When multi-intent detected, store proposal for task_decomposer.
    """
    # Get the latest message
    messages = state.get("messages", [])
    if not messages:
        return {}

    latest = messages[-1].content if messages else ""

    # Classify intent (now returns TaskPlanProposal)
    from src.graph.intent import classify_intent

    proposal = await classify_intent(
        message=latest,
        context=state.get("conversation_context"),
        thread_context=state.get("thread_context"),
        return_proposal=True,  # New flag for multi-intent
    )

    # Store proposal in intent_result for downstream use
    intent_result = proposal.to_single_intent().model_dump()
    intent_result["task_plan_proposal"] = proposal.model_dump()

    # Set super_mode for UI
    intent_result["super_mode"] = proposal.primary_mode.value

    return {
        "intent_result": intent_result,
    }
```
