def build_used_attachments_footer(
    context: "AttachmentContext",
) -> list[dict]:
    """Build footer showing what attachments were used.

    Displays:
    - 📎 Used: filename (sections: 1, 2, 3)
    - [Show sources] [Stop using this file]

    Args:
        context: AttachmentContext from retriever.

    Returns:
        List of Block Kit blocks for response footer.
    """
    if not context.pinned and not context.retrieved_chunks:
        return []

    blocks = []

    # Build "Used" summary
    used_files = {}

    # Collect pinned files
    for p in context.pinned:
        filename = p["filename"]
        if filename not in used_files:
            used_files[filename] = {
                "id": p["id"],
                "sections": [],
                "pinned": True,
            }

    # Collect retrieved chunks
    for c in context.retrieved_chunks:
        filename = c["filename"]
        if filename not in used_files:
            used_files[filename] = {
                "id": c["attachment_id"],
                "sections": [],
                "pinned": False,
            }
        used_files[filename]["sections"].append(c["chunk_index"])

    # Build text
    used_parts = []
    for filename, info in used_files.items():
        part = f"*{filename}*"
        if info["pinned"]:
            part += " (📌 pinned)"
        elif info["sections"]:
            sections = sorted(set(info["sections"]))[:5]
            part += f" (sections: {', '.join(map(str, sections))})"
        used_parts.append(part)

    used_text = "📎 Used: " + ", ".join(used_parts)

    blocks.append({
        "type": "context",
        "elements": [
            {"type": "mrkdwn", "text": used_text}
        ],
    })

    # Add action buttons if we have used files
    if used_files:
        actions = []

        # Show sources button (if we have retrieved chunks)
        if context.retrieved_chunks:
            actions.append({
                "type": "button",
                "text": {"type": "plain_text", "text": "Show sources"},
                "action_id": "attachment_show_sources",
                "value": _encode_source_ids(context),
            })

        # Stop using button (for each file)
        for filename, info in list(used_files.items())[:2]:
            actions.append({
                "type": "button",
                "text": {
                    "type": "plain_text",
                    "text": f"Stop using {filename[:15]}..."
                            if len(filename) > 15 else f"Stop using {filename}",
                },
                "action_id": "attachment_stop_using",
                "value": info["id"],
            })

        if actions:
            blocks.append({
                "type": "actions",
                "elements": actions,
            })

    return blocks


def build_sources_modal(
    context: "AttachmentContext",
) -> dict:
    """Build modal showing source content.

    Displays the actual chunks that were used.

    Args:
        context: AttachmentContext with retrieved chunks.

    Returns:
        Slack modal view payload.
    """
    blocks = []

    # Group chunks by file
    chunks_by_file = {}
    for c in context.retrieved_chunks:
        filename = c["filename"]
        if filename not in chunks_by_file:
            chunks_by_file[filename] = []
        chunks_by_file[filename].append(c)

    # Build blocks for each file
    for filename, chunks in chunks_by_file.items():
        blocks.append({
            "type": "header",
            "text": {"type": "plain_text", "text": f"📄 {filename}"},
        })

        for chunk in chunks:
            # Truncate content for display
            content = chunk["content"][:1000]
            if len(chunk["content"]) > 1000:
                content += "..."

            blocks.append({
                "type": "section",
                "text": {
                    "type": "mrkdwn",
                    "text": f"*Section {chunk['chunk_index']}* (relevance: {chunk.get('score', 0):.2f})\n```{content}```",
                },
            })

        blocks.append({"type": "divider"})

    return {
        "type": "modal",
        "title": {"type": "plain_text", "text": "Source Content"},
        "close": {"type": "plain_text", "text": "Close"},
        "blocks": blocks[:50],  # Slack limit
    }


def build_offer_attachments_block(
    context: "AttachmentContext",
) -> Optional[dict]:
    """Build block offering to use available attachments.

    For CHAT mode when attachments exist but aren't included.

    Args:
        context: AttachmentContext with offer_available.

    Returns:
        Block or None if no attachments to offer.
    """
    offer_msg = context.to_offer_message()
    if not offer_msg:
        return None

    return {
        "type": "section",
        "text": {"type": "mrkdwn", "text": offer_msg},
    }


def _encode_source_ids(context: "AttachmentContext") -> str:
    """Encode source IDs for button value.

    Slack button values have length limits.
    """
    import json
    ids = [c["attachment_id"] for c in context.retrieved_chunks[:5]]
    return json.dumps({"ids": ids, "query": ""})[:2000]

async def handle_show_sources(
    client: AsyncWebClient,
    body: dict,
) -> None:
    """Handle Show Sources button click.

    Opens modal with source content.
    """
    import json
    from src.slack.blocks.attachments import build_sources_modal
    from src.documents.retriever import AttachmentContext

    trigger_id = body.get("trigger_id")
    action = body.get("actions", [{}])[0]
    value = action.get("value", "{}")

    try:
        data = json.loads(value)
        ids = data.get("ids", [])
    except json.JSONDecodeError:
        return

    # Rebuild context from IDs (simplified for modal)
    # In practice, you might cache the context or rebuild from chunks
    context = AttachmentContext()

    async with get_connection() as conn:
        chunk_store = AttachmentChunkStore(conn)
        store = AttachmentStore(conn)

        for attachment_id in ids:
            chunks = await chunk_store.get_chunks(UUID(attachment_id), limit=5)
            attachment = await store.get(UUID(attachment_id))

            for chunk in chunks:
                context.retrieved_chunks.append({
                    "attachment_id": str(chunk.attachment_id),
                    "filename": attachment.filename if attachment else "unknown",
                    "chunk_index": chunk.chunk_index,
                    "content": chunk.chunk_text,
                    "score": 0,
                })

    modal = build_sources_modal(context)

    await client.views_open(
        trigger_id=trigger_id,
        view=modal,
    )


async def handle_stop_using(
    client: AsyncWebClient,
    body: dict,
    respond: Callable,
) -> None:
    """Handle Stop Using button click.

    Unpins the attachment.
    """
    action = body.get("actions", [{}])[0]
    attachment_id = action.get("value")

    if not attachment_id:
        return

    # Delegate to unpin
    await handle_attachment_unpin(client, {
        "user": body.get("user"),
        "actions": [{"value": attachment_id}],
    }, respond)


# Update register_attachment_handlers to include new handlers:
def register_attachment_handlers(app) -> None:
    """Register all attachment button handlers."""

    @app.action("attachment_pin")
    async def on_pin(ack, body, client, respond):
        await ack()
        await handle_attachment_pin(client, body, respond)

    @app.action("attachment_unpin")
    async def on_unpin(ack, body, client, respond):
        await ack()
        await handle_attachment_unpin(client, body, respond)

    @app.action("attachment_show_sources")
    async def on_show_sources(ack, body, client):
        await ack()
        await handle_show_sources(client, body)

    @app.action("attachment_stop_using")
    async def on_stop_using(ack, body, client, respond):
        await ack()
        await handle_stop_using(client, body, respond)

from src.slack.blocks.attachments import (
    build_used_attachments_footer,
    build_offer_attachments_block,
)
from src.documents.retriever import AttachmentContext

async def post_response(
    client: AsyncWebClient,
    channel_id: str,
    thread_ts: Optional[str],
    text: str,
    blocks: list[dict],
    attachment_context: Optional[AttachmentContext] = None,
) -> dict:
    """Post bot response with transparency footer.

    Adds attachment usage info to response.
    """
    final_blocks = list(blocks) if blocks else []

    # Add attachment transparency footer
    if attachment_context:
        # Add "used" footer if content was included
        footer = build_used_attachments_footer(attachment_context)
        if footer:
            final_blocks.extend(footer)

        # Or add "offer" block if in CHAT mode
        offer = build_offer_attachments_block(attachment_context)
        if offer:
            final_blocks.append(offer)

    return await client.chat_postMessage(
        channel=channel_id,
        thread_ts=thread_ts,
        text=text,
        blocks=final_blocks if final_blocks else None,
    )
