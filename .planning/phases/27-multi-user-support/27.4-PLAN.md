# Phase 27.4: State-Bound Approvals

**Requirements:** R13 (State-bound approvals), R14 (Configurable policy), R15 (First-wins)
**Risk:** Medium
**Dependencies:** 27.1 (User Identity)

---

## Objective

Tie approvals to exact state version to prevent approving outdated content. Implement configurable approval policies (any_contributor, only_admins, two_person). First approval wins and replaces buttons with resolution text.

**Key behaviors:**
- Approval buttons include `state_version` in payload
- Outdated approval → "This preview is outdated, please review the updated version"
- First click wins, subsequent clicks → "Already approved by @user"
- Configurable policy per channel
- Two-person rule: creator cannot approve their own work

```python
from enum import Enum
from typing import Optional
from pydantic import BaseModel, Field


class ApprovalPolicy(str, Enum):
    """Configurable approval policies for channels."""
    ANY_CONTRIBUTOR = "any_contributor"  # Anyone in thread can approve
    ONLY_ADMINS = "only_admins"  # Only channel admins can approve
    TWO_PERSON = "two_person"  # Creator cannot approve their own work
    ROLE_BASED = "role_based"  # Specific roles can approve specific types


class ApprovalRequirement(BaseModel):
    """Approval requirement for a specific action."""
    action_type: str = Field(description="Type of action: jira_create, jira_update, etc.")
    policy: ApprovalPolicy = Field(default=ApprovalPolicy.ANY_CONTRIBUTOR)
    required_role: Optional[str] = Field(default=None, description="For ROLE_BASED policy")
    min_approvers: int = Field(default=1, description="Minimum approvers needed")


class ChannelApprovalConfig(BaseModel):
    """Approval configuration for a channel."""
    channel_id: str
    default_policy: ApprovalPolicy = Field(default=ApprovalPolicy.ANY_CONTRIBUTOR)
    requirements: list[ApprovalRequirement] = Field(default_factory=list)
    created_by: str
    updated_at: str
```

### 3. Approval Policy Store (NEW)

**File:** `src/db/approval_policy_store.py`

```python
from datetime import datetime, timezone
from typing import Optional
import json
from psycopg import AsyncConnection

from src.schemas.approval import ApprovalPolicy, ChannelApprovalConfig


class ApprovalPolicyStore:
    """Store channel approval policies."""

    def __init__(self, conn: AsyncConnection) -> None:
        self._conn = conn

    async def ensure_table(self) -> None:
        """Create approval_policies table."""
        async with self._conn.cursor() as cur:
            await cur.execute("""
                CREATE TABLE IF NOT EXISTS approval_policies (
                    channel_id TEXT PRIMARY KEY,
                    default_policy TEXT NOT NULL DEFAULT 'any_contributor',
                    requirements JSONB DEFAULT '[]',
                    created_by TEXT NOT NULL,
                    created_at TIMESTAMPTZ DEFAULT NOW(),
                    updated_at TIMESTAMPTZ DEFAULT NOW()
                )
            """)
            await self._conn.commit()

    async def get_policy(self, channel_id: str) -> Optional[ChannelApprovalConfig]:
        """Get approval policy for channel."""
        async with self._conn.cursor() as cur:
            await cur.execute("""
                SELECT * FROM approval_policies WHERE channel_id = %s
            """, (channel_id,))
            row = await cur.fetchone()

        if not row:
            return None

        return ChannelApprovalConfig(
            channel_id=row[0],
            default_policy=ApprovalPolicy(row[1]),
            requirements=row[2] or [],
            created_by=row[3],
            updated_at=row[5].isoformat(),
        )

    async def set_policy(
        self,
        channel_id: str,
        policy: ApprovalPolicy,
        user_id: str,
    ) -> ChannelApprovalConfig:
        """Set default approval policy for channel."""
        now = datetime.now(timezone.utc)
        async with self._conn.cursor() as cur:
            await cur.execute("""
                INSERT INTO approval_policies (channel_id, default_policy, created_by, updated_at)
                VALUES (%s, %s, %s, %s)
                ON CONFLICT (channel_id) DO UPDATE SET
                    default_policy = EXCLUDED.default_policy,
                    updated_at = EXCLUDED.updated_at
                RETURNING *
            """, (channel_id, policy.value, user_id, now))
            row = await cur.fetchone()
            await self._conn.commit()

        return ChannelApprovalConfig(
            channel_id=row[0],
            default_policy=ApprovalPolicy(row[1]),
            requirements=row[2] or [],
            created_by=row[3],
            updated_at=row[5].isoformat(),
        )

    async def get_effective_policy(
        self,
        channel_id: str,
        action_type: str = "jira_create",
    ) -> ApprovalPolicy:
        """Get effective policy for an action, falling back to defaults."""
        config = await self.get_policy(channel_id)
        if not config:
            return ApprovalPolicy.ANY_CONTRIBUTOR

        # Check for specific requirement
        for req in config.requirements:
            if req.get("action_type") == action_type:
                return ApprovalPolicy(req.get("policy", config.default_policy.value))

        return config.default_policy
```

### 4. Approval Validator (NEW)

**File:** `src/slack/approval_validator.py`

```python
from typing import Optional, Tuple
import logging

from src.schemas.approval import ApprovalPolicy
from src.db import get_connection
from src.db.approval_policy_store import ApprovalPolicyStore

logger = logging.getLogger(__name__)


async def validate_approval(
    channel_id: str,
    user_id: str,
    creator_user_id: str,
    state_version: int,
    current_state_version: int,
    draft_hash: str,
    action_type: str = "jira_create",
) -> Tuple[bool, Optional[str]]:
    """Validate if user can approve this action.

    Returns:
        (is_valid, error_message) — If invalid, error_message explains why
    """
    # Check state version (outdated approval)
    if state_version != current_state_version:
        return False, "This preview is outdated. Please review the updated version."

    # Get channel policy
    async with get_connection() as conn:
        store = ApprovalPolicyStore(conn)
        policy = await store.get_effective_policy(channel_id, action_type)

    # Validate based on policy
    if policy == ApprovalPolicy.TWO_PERSON:
        if user_id == creator_user_id:
            return False, "Two-person rule: You cannot approve your own work. Please ask another team member."

    if policy == ApprovalPolicy.ONLY_ADMINS:
        # TODO: Check if user is channel admin via Slack API
        # For now, allow (would need Slack API integration)
        pass

    return True, None


async def check_already_approved(
    session_id: str,
    draft_hash: str,
) -> Tuple[bool, Optional[str]]:
    """Check if this draft was already approved.

    Returns:
        (already_approved, approver_display) — If approved, who approved it
    """
    from src.db.approval_store import ApprovalStore
    from src.db.user_metadata_store import UserMetadataStore

    async with get_connection() as conn:
        approval_store = ApprovalStore(conn)
        existing = await approval_store.get_approval(session_id, draft_hash)

        if not existing:
            return False, None

        # Get approver display name
        user_store = UserMetadataStore(conn)
        display_name = await user_store.get_display_name(
            existing.approved_by,
            default=f"<@{existing.approved_by}>"
        )

        return True, display_name
```

```python
async def _handle_approve_async(body, client, action):
    import json

    channel_id = body["channel"]["id"]
    thread_ts = body["message"].get("thread_ts") or body["message"]["ts"]
    user_id = body["user"]["id"]

    # Parse button value
    value = json.loads(action.get("value", "{}"))
    draft_hash = value.get("draft_hash")
    button_state_version = value.get("state_version", 0)

    # Get current state
    from src.graph.runner import get_runner
    from src.slack.session import SessionIdentity

    identity = SessionIdentity(team_id="", channel_id=channel_id, thread_ts=thread_ts)
    runner = get_runner(identity)
    state = await runner._get_current_state()

    current_state_version = state.get("state_version", 0)
    creator_user_id = state.get("user_id", "")

    # Validate approval
    from src.slack.approval_validator import validate_approval, check_already_approved

    # Check if already approved
    session_id = identity.session_id
    already_approved, approver = await check_already_approved(session_id, draft_hash)
    if already_approved:
        client.chat_postEphemeral(
            channel=channel_id,
            user=user_id,
            text=f"Already approved by {approver}",
        )
        return

    # Validate this approval
    is_valid, error = await validate_approval(
        channel_id=channel_id,
        user_id=user_id,
        creator_user_id=creator_user_id,
        state_version=button_state_version,
        current_state_version=current_state_version,
        draft_hash=draft_hash,
    )

    if not is_valid:
        client.chat_postEphemeral(
            channel=channel_id,
            user=user_id,
            text=error,
        )
        return

    # Record approval with state binding
    from src.db import get_connection
    from src.db.approval_store import ApprovalStore

    async with get_connection() as conn:
        store = ApprovalStore(conn)
        is_new = await store.record_approval(
            session_id=session_id,
            draft_hash=draft_hash,
            approved_by=user_id,
            state_version=current_state_version,
            ui_version=state.get("ui_version", 0),
        )

    if not is_new:
        # Race condition - someone else approved first
        _, approver = await check_already_approved(session_id, draft_hash)
        client.chat_postEphemeral(
            channel=channel_id,
            user=user_id,
            text=f"Already approved by {approver}",
        )
        return

    # Proceed with approval...
    # (existing approval logic)
```
